HAS_neidran_merc_contract = {
	icon = GFX_focus_ETH_keep_the_chitet
	
	enable = {  
		always = yes
	}
	
	# 如果和奈德兰 (NEI) 开战，修饰符自动失效并移除！
	remove_trigger = {
		has_war_with = NED
	}
	
	# 内战叛军会继承这个修饰符（以及相关的负债）
	allowed_civil_war = { always = yes }
	
	# 添加修饰符时，强制重新算账，防止数值出错
	on_add = {
		HAS_calculate_merc_cost_effect = yes
	}
	
	# 被移除时
	on_remove = {
		# 解散所有佣兵，且不返还装备 (disband = no)
		delete_units = { division_template = "奈德兰佣兵团" disband = no }
		set_variable = { HAS_neidran_merc_cost = 0 }
		clr_country_flag = HAS_has_active_merc_contract
	}
	
	# 扣除民用工厂，数量 = HAS_neidran_merc_cost 变量
	civilian_factory_use = HAS_neidran_merc_cost
	special_forces_min = HAS_neidran_merc_special_forces_min 
}
