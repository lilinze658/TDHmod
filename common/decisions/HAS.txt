HAS_mercenary_contracts_category = {

	HAS_merc_contract_timer_mission = {
		icon = generic_economic_treaty
		is_good = yes
		days_mission_timeout = 180
		
		# 当拥有活跃合同时，进度条出现并激活
		activation = {
			always = no
		}
		
		# 中断条件：如果我们与奈德兰 (NEI) 开战了，进度条立刻中断并消失！
		cancel_trigger = {
			has_war_with = NEI
		}
		
		# 当因开战而被迫中断时，执行以下效果：强行解散佣兵，取消工厂扣除
		# (这就替代了动态修饰符里的 remove_trigger 功能，且更直观)
		on_fail = {
			delete_units = { division_template = "奈德兰佣兵团" disband = no } # 人枪全收
			set_variable = { HAS_neidran_merc_cost = 0 }
			set_variable = { HAS_merc_divisions_contracted = 0 }
			remove_dynamic_modifier = { modifier = HAS_neidran_merc_contract }
			clr_country_flag = HAS_has_active_merc_contract
		}

		# 正常倒计时结束 (180天到期)：触发隐藏算账事件！
		timeout_effect = {
			country_event = { id = HAS_mercenary.100 }
		}
	}

	HAS_hire_neidran_security = {
		icon = generic_army_motorized
		
		# 【可见条件】：当前没有合同
		visible = {
			has_country_flag = HAS_can_hire_neidran_mercs
			NOT = { has_country_flag = HAS_has_active_merc_contract }
		}
		

		available = {
			NOT = { has_war_with = NEI }
			num_of_civilian_factories_available_for_projects > 10 
		}
		
		cost = 10
		complete_effect = {
			country_event = { id = HAS_mercenary.10 } # 弹出初次雇佣事件
		}

	}

	# 2. 追加佣兵雇佣 (已有合同时)
	HAS_hire_more_neidran_mercs = {
		icon = generic_production_bonus
		visible = {
			has_country_flag = HAS_has_active_merc_contract
		}
		available = {
			NOT = { has_war_with = NEI }
			num_of_civilian_factories_available_for_projects > 10 
		}
		cost = 0
		complete_effect = {
			country_event = { id = HAS_mercenary.11 } # 弹出加购事件
		}
	}

	# 3. 提前遣散佣兵
	HAS_dismiss_neidran_mercs_early = {
		icon = generic_cancel_deal
		# 【可见条件】：当前已有合同
		visible = {
			has_country_flag = HAS_has_active_merc_contract
		}
		
		available = {
			# 确保地图上还有活着的佣兵可以遣散，不然这个按钮没意义
			any_country_division = { division_template = "奈德兰佣兵团" }
		}
		cost = 0
		complete_effect = {
			delete_units = { division_template = "奈德兰佣兵团" disband = no }###不返还人力装备
			# 注意！我们【不】调用算账脚本，也【不】移除动态修饰符！
			# 这意味着 HAS_neidran_merc_cost 变量保持原样，工厂继续扣除。
			# 直到 180 天的到期事件触发时，底层脚本才会因为找不到部队而把花费清零。
			
			custom_effect_tooltip = HAS_dismiss_neidran_mercs_early_tt
		}
	}


	HAS_hire_neidran_merc_regiment = {
