# 重新计算佣兵费用的脚本
HAS_calculate_merc_cost_effect = {
	# 1. 先把佣兵数量变量清零
	set_variable = { HAS_merc_divisions_contracted = 0 }
	
	# 2. 扫描全国所有师团
	every_country_division = {
		limit = {
			has_template = "奈德兰佣兵团" # 检查模板名称
		}
		# 找到一个，就给国家的变量 +1 (PREV代表退回国家作用域)
		PREV = { add_to_variable = { HAS_merc_divisions_contracted = 1 } }
	}
	
	# 3. 计算工厂花费 (数量 * 3)
	set_variable = { HAS_merc_civ_cost = HAS_merc_divisions_contracted }
	multiply_variable = { HAS_merc_civ_cost = 3 }
	
	# 4. 确保动态修饰符挂在身上
	#add_dynamic_modifier = { modifier = HAS_neidran_merc_contract_cost }
}

# 根据 event_target:ideology_source 的子政体，给当前国家 (ROOT) 挂载对应的 Buff
apply_subideology_buffs_from_source_effect = {


    # 2. 同步子政体并挂载新 Buff
    if = {
        limit = { event_target:ideology_source = { has_country_leader_ideology = religious_conservatism } }
        set_country_leader_ideology = religious_conservatism
        add_dynamic_modifier = { modifier = subideology_religious_conservatism_modifier }
        add_ideas = subideology_religious_conservatism_modifier_rule
    }
    else_if = {
        limit = { event_target:ideology_source = { has_country_leader_ideology = social_conservatism } }
        set_country_leader_ideology = social_conservatism
        add_dynamic_modifier = { modifier = subideology_social_conservatism_modifier }
        add_ideas = subideology_social_conservatism_modifier_rule
    }
    else_if = {
        limit = { event_target:ideology_source = { has_country_leader_ideology = aristocratic_republicanism } }
        set_country_leader_ideology = aristocratic_republicanism
        add_dynamic_modifier = { modifier = subideology_aristocratic_republicanism_modifier }
        add_ideas = subideology_aristocratic_republicanism_modifier_rule
    }
    else_if = {
        limit = { event_target:ideology_source = { has_country_leader_ideology = dynastic_legitimism } }
        set_country_leader_ideology = dynastic_legitimism
        add_dynamic_modifier = { modifier = subideology_dynastic_legitimism_modifier }
        add_ideas = subideology_dynastic_legitimism_modifier_rule
    }
}

apply_own_subideology_buffs_effect = {

    # --- 第一步：清除所有旧的子意识形态修正 ---
    if = { limit = { has_dynamic_modifier = { modifier = subideology_religious_conservatism_modifier } } remove_dynamic_modifier = { modifier = subideology_religious_conservatism_modifier } }
    if = { limit = { has_dynamic_modifier = { modifier = subideology_social_conservatism_modifier } } remove_dynamic_modifier = { modifier = subideology_social_conservatism_modifier } }
    if = { limit = { has_dynamic_modifier = { modifier = subideology_aristocratic_republicanism_modifier } } remove_dynamic_modifier = { modifier = subideology_aristocratic_republicanism_modifier } }
    if = { limit = { has_dynamic_modifier = { modifier = subideology_dynastic_legitimism_modifier } } remove_dynamic_modifier = { modifier = subideology_dynastic_legitimism_modifier } }

    if = { limit = { has_idea = subideology_religious_conservatism_modifier_rule } remove_ideas = subideology_religious_conservatism_modifier_rule }
    if = { limit = { has_idea = subideology_social_conservatism_modifier_rule } remove_ideas = subideology_social_conservatism_modifier_rule }
    if = { limit = { has_idea = subideology_aristocratic_republicanism_modifier_rule } remove_ideas = subideology_aristocratic_republicanism_modifier_rule }
    if = { limit = { has_idea = subideology_dynastic_legitimism_modifier_rule } remove_ideas = subideology_dynastic_legitimism_modifier_rule }

    # --- 第二步：读取自身当前的子意识形态，挂载对应的新 Buff ---
    if = {
        limit = { has_country_leader_ideology = religious_conservatism }
        add_dynamic_modifier = { modifier = subideology_religious_conservatism_modifier }
        add_ideas = subideology_religious_conservatism_modifier_rule
    }
    else_if = {
        limit = { has_country_leader_ideology = social_conservatism }
        add_dynamic_modifier = { modifier = subideology_social_conservatism_modifier }
        add_ideas = subideology_social_conservatism_modifier_rule
    }
    else_if = {
        limit = { has_country_leader_ideology = aristocratic_republicanism }
        add_dynamic_modifier = { modifier = subideology_aristocratic_republicanism_modifier }
        add_ideas = subideology_aristocratic_republicanism_modifier_rule
    }
    else_if = {
        limit = { has_country_leader_ideology = dynastic_legitimism }
        add_dynamic_modifier = { modifier = subideology_dynastic_legitimism_modifier }
        add_ideas = subideology_dynastic_legitimism_modifier_rule
    }
}

# 将当前国家 (ROOT) 的子意识形态同步为宗主国的
# 前提：ROOT 必须是傀儡国 (is_subject = yes)
# ==========================================
sync_subject_ideology_to_overlord_effect = {

    # ---------- 宗教保守主义 ----------
    if = {
        limit = {
            overlord = { has_country_leader_ideology = religious_conservatism }
            NOT = { has_country_leader_ideology = religious_conservatism }
        }
        # 国内有人选？
        if = {
            limit = {
                any_character = {
                    can_be_country_leader = yes
                    has_ideology = religious_conservatism
                }
            }
            random_character = {
                limit = {
                    can_be_country_leader = yes
                    has_ideology = religious_conservatism
                }
                promote_character = { ideology = religious_conservatism }
            }
        }
        # 没有人选，捏大众脸
        else = {
            create_country_leader = {
                ideology = religious_conservatism
            }
        }
    }

    # ---------- 社会保守主义 ----------
    else_if = {
        limit = {
            overlord = { has_country_leader_ideology = social_conservatism }
            NOT = { has_country_leader_ideology = social_conservatism }
        }
        if = {
            limit = {
                any_character = {
                    can_be_country_leader = yes
                    has_ideology = social_conservatism
                }
            }
            random_character = {
                limit = {
                    can_be_country_leader = yes
                    has_ideology = social_conservatism
                }
                promote_character = { ideology = social_conservatism }
            }
        }
        else = {
            create_country_leader = {
                name = "Puppet Government"
                ideology = social_conservatism
                traits = { }
            }
        }
    }

    # ---------- 贵族共和主义 ----------
    else_if = {
        limit = {
            overlord = { has_country_leader_ideology = aristocratic_republicanism }
            NOT = { has_country_leader_ideology = aristocratic_republicanism }
        }
        if = {
            limit = {
                any_character = {
                    can_be_country_leader = yes
                    has_ideology = aristocratic_republicanism
                }
            }
            random_character = {
                limit = {
                    can_be_country_leader = yes
                    has_ideology = aristocratic_republicanism
                }
                promote_character = { ideology = aristocratic_republicanism }
            }
        }
        else = {
            create_country_leader = {
                name = "Puppet Government"
                ideology = aristocratic_republicanism
                traits = { }
            }
        }
    }

    # ---------- 王朝正统主义 ----------
    else_if = {
        limit = {
            overlord = { has_country_leader_ideology = dynastic_legitimism }
            NOT = { has_country_leader_ideology = dynastic_legitimism }
        }
        if = {
            limit = {
                any_character = {
                    can_be_country_leader = yes
                    has_ideology = dynastic_legitimism
                }
            }
            random_character = {
                limit = {
                    can_be_country_leader = yes
                    has_ideology = dynastic_legitimism
                }
                promote_character = { ideology = dynastic_legitimism }
            }
        }
        else = {
            create_country_leader = {
                name = "Puppet Government"
                ideology = dynastic_legitimism
                traits = { }
            }
        }
    }

    # ---------- 最后：调用万能脚本刷新 Buff ----------
    apply_own_subideology_buffs_effect = yes
}
