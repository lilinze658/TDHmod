# 加安斯收到哈斯佩芙的外交提议
country_event = {
    id = HAS_diplomacy.1
    title = "HAS_diplomacy.1.t"
    desc = "HAS_diplomacy.1.d"
    picture = GFX_report_event_generic_sign_treaty
    
    is_triggered_only = yes

    # 选项A：接受提议
    option = {
        name = "HAS_diplomacy.1.a"
        ai_chance = { factor = 99 } # AI大概率接受
        
        # 加安斯对哈斯佩芙好感+30
        add_opinion_modifier = { target = HAS modifier = HAS_diplomatic_ties_gaans }
        
        # 哈斯佩芙对加安斯好感+30，并获得国家精神
        HAS = {
            add_opinion_modifier = { target = REP modifier = HAS_diplomatic_ties_gaans }
            add_opinion_modifier = { target = REP modifier = HAS_mutual_pact_trade_bias }
            add_ideas = HAS_gaans_agri_quota
            country_event = { id = HAS_diplomacy.2 } # 发送成功通知给哈斯佩芙
            give_market_access = REP
        }
    }

    # 选项B：拒绝
    option = {
        name = "HAS_diplomacy.1.b"
        ai_chance = { factor = 10 }
        
        HAS = {
            country_event = { id = HAS_diplomacy.3 } # 发送失败通知
        }
    }
}
