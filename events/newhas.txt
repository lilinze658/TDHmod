# 加安斯收到哈斯佩芙的外交提议
country_event = {
    id = HAS_diplomacy.1
    title = "HAS_diplomacy.1.t"
    desc = "HAS_diplomacy.1.d"
    picture = GFX_report_event_generic_sign_treaty
    
    is_triggered_only = yes

    # 选项A：接受提议
    option = {
        name = "HAS_diplomacy.1.a"
        ai_chance = { factor = 90 } # AI大概率接受
        
        # 加安斯对哈斯佩芙好感+30
        add_opinion_modifier = { target = HAS modifier = HAS_diplomatic_ties_gaans }
        
        # 哈斯佩芙对加安斯好感+30，并获得国家精神
        HAS = {
            add_opinion_modifier = { target = REP modifier = HAS_diplomatic_ties_gaans }
            add_opinion_modifier = { target = REP modifier = HAS_mutual_pact_trade_bias }
            add_ideas = HAS_gaans_agri_quota
            country_event = { id = HAS_diplomacy.2 } # 发送成功通知给哈斯佩芙
        }
    }

    # 选项B：拒绝
    option = {
        name = "HAS_diplomacy.1.b"
        ai_chance = { factor = 10 }
        
        HAS = {
            country_event = { id = HAS_diplomacy.3 } # 发送失败通知
        }
    }
}




# 决定用多大精力学习的事件
country_event = {
    id = HAS_saint.1
    title = "HAS_saint.1.t"
    desc = "HAS_saint.1.desc"
    picture = GFX_report_event_generic_read_write
    is_triggered_only = yes


    # 选项 A：本圣女是天才！(耗时极长，花费少)
    option = {
        name = "HAS_saint.1.a"
        add_political_power = -10
        
        # 启动读条
        set_country_flag = HAS_saint_is_studying_flag
        activate_mission = HAS_saint_is_studying_mission
        add_days_mission_timeout = { mission = HAS_saint_is_studying_mission days = 120 } # 总计 120 天
    }


    # 选项 B：努力去学习 (适中)
    option = {
        name = "HAS_saint.1.b"
        trigger = { has_political_power > 29 }
        add_political_power = -30
        add_command_power = -10 # 消耗一点指挥点代表精力
        
        set_country_flag = HAS_saint_is_studying_flag
        activate_mission = HAS_saint_is_studying_mission
        add_days_mission_timeout = { mission = HAS_saint_is_studying_mission days = 60 } # 总计 60 天
    }


    # 选项 C：尽全力去学习！(极速，极贵)
    option = {
        name = "HAS_saint.1.c"
        trigger = { has_political_power > 74 }
        add_political_power = -75
        add_command_power = -30
        
        set_country_flag = HAS_saint_is_studying_flag
        activate_mission = HAS_saint_is_studying_mission
        add_days_mission_timeout = { mission = HAS_saint_is_studying_mission days = 3 } # 总计 30 天
    }


    # 选项 D：还是算了
    option = {
        name = "HAS_saint.1.d"
        # 什么都不发生，退回决议
    }
}
